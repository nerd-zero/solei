## Quickstart

```go icon="golang" main.go
package main

import (
	"context"
	soleigo "github.com/nerd-zero/solei-go"
	"log"
	"os"
)

func main() {
	ctx := context.Background()

	s := soleigo.New(
		soleigo.WithSecurity(os.Getenv("SOLEI_ACCESS_TOKEN")),
	)

	res, err := s.Organizations.List(ctx, nil, soleigo.Pointer[int64](1), soleigo.Pointer[int64](10), nil)
	if err != nil {
		log.Fatal(err)
	}
	if res.ListResourceOrganization != nil {
		for {
			// handle items

			res, err = res.Next()

			if err != nil {
				// handle error
			}

			if res == nil {
				break
			}
		}
	}
}

```
